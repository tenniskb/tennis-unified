<script>
(function {
  var path = window.location.pathname;
  var search = window.location.search;
  var hash = window.location.hash;

  // 1. If path ends with .md or .md/
  if (path.endsWith('.md')) {
    var target = path.slice(0, -3) + '.html';
    window.location.replace(target + search + hash);
    return;
  }
  if (path.endsWith('.md/')) {
    var target = path.slice(0, -4) + '.html';
    window.location.replace(target + search + hash);
    return;
  }

  // 2. If path has no extension and does not end with /
  if (!path.endsWith('.html') && !path.endsWith('/') && path.indexOf('.') === -1) {
    var target = path + '.html';
    window.location.replace(target + search + hash);
    return;
  }
});
</script>
